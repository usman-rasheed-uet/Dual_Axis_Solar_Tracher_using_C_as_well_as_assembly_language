    
OVER CALL DELAY
    
     BSF ADCON0, GO
    
BACK BTFSC ADCON0,DONE
     BRA BACK
     
     MOVFF ADRESL,L_BYTE
     MOVFF ADRESH,H_BYTE
     
     CALL ALGO_10_TO_8
     
    ; MOVFF BIN_TEMP,PORTD
     
     BRA OVER
     
ALGO_10_TO_8
     
     RRNCF L_BYTE,F
     RRNCF L_BYTE,W
     ANDLW 0X3F
     MOVWF L_BYTE
     RRNCF H_BYTE,F
     RRNCF H_BYTE,W
     ANDLW 0XC0
     IORWF L_BYTE,W
     MOVWF BIN_TEMP
     RETURN
DELAY
     MOVLW D'3'
     MOVWF EQ
AGAIN   DECF EQ,F
     BNZ AGAIN

;PROGRAM FOR 2ND LDR2 
     
L_BYTE2 SET 0x40
H_BYTE2 SET 0x41
BIN_TEMP2 SET 0x42
 
    ;CLRF TRISC
    BSF TRISA,RA1
    
    MOVLW 0x89
    MOVWF ADCON0
    
    MOVLW 0xC4
    MOVWF ADCON1
    
OVER2 CALL DELAY2
    
     BSF ADCON0, GO
    
BACK2 BTFSC ADCON0,DONE
     BRA BACK2
     
     MOVFF ADRESL,L_BYTE2
     MOVFF ADRESH,H_BYTE2
     
     CALL ALGO_10_TO_82
     
    ; MOVFF BIN_TEMP2,PORTC
     
     BRA OVER2
     
ALGO_10_TO_82
     
     RRNCF L_BYTE2,F
     RRNCF L_BYTE2,W
     ANDLW 0X3F
     MOVWF L_BYTE2
     RRNCF H_BYTE2,F
     RRNCF H_BYTE2,W
     ANDLW 0XC0
     IORWF L_BYTE2,W
     MOVWF BIN_TEMP2
     RETURN
DELAY2
     MOVLW D'3'
     MOVWF EQ2
AGAIN2   DECF EQ2,F
     BNZ AGAIN2
     
;COMPARE VALUES OF SENSORS
     
;BIN_TEMP2 SET 0x50
     
	  MOVFF BIN_TEMP,W
LDR1_CHECK     
	  CPFSGT BIN_TEMP2
	  BRA LDR2_CHECK
	  CALL R_CLOCKW
	  BRA LDR1_CHECK
	  
LDR2_CHECK
	  CPFSLT  BIN_TEMP2
	  BRA LDR1_LDR2
	  CALL R_ANTICLOCKW
	  BRA LDR2_CHECK
	  
LDR1_LDR2
	  CPFSEQ BIN_TEMP2
	  BRA LDR1_CHECK
	  GOTO NO_MOTION
	  BRA LDR1_LDR2
	  
;STEPPER MOTOR ROTATION
	  
	  
	  
AB EQU 0x60
MOTVALUE SET 0x61
EMPVALUE SET 0x62    
 
     MOVLW 0x66
     MOVWF MOTVALUE

R_CLOCKW
H	MOVFF MOTVALUE,PORTB
	RRNCF MOTVALUE,F
	CALL DELAY3
	GOTO H
	;BRA LDR1_CHECK
	RETURN
R_ANTICLOCKW 
G	MOVFF MOTVALUE,PORTB
	RLNCF MOTVALUE,F
	CALL DELAY3
	GOTO G
	;BRA LDR2_CHECK
	RETURN
NO_MOTION
I	MOVLW 0x00
	MOVWF EMPVALUE
	MOVFF EMPVALUE,PORTB
	CALL DELAY3
	GOTO I
	
	;BRA LDR1_LDR2
      RETURN
DELAY3
      
    MOVLW D'255'
    MOVWF AB
      
THERE

    NOP
      DECF AB,F
