#include "P18f452.inc"
    LIST P=18f452, F=INHX32, MM=OFF
    CONFIG OSC=XT
    CONFIG WDT=OFF ORG 0x00
   
AB EQU 0x30
STARTREG SET 0x40
STOPREG SET 0x50
    GOTO MAIN
MAIN
    BSF TRISD, RD7
    CLRF TRISB
	MOVLW 0x66
	MOVWF STARTREG
	MOVLW 0x00
	MOVWF STOPREG
 
BACK BTFSS PORTD, RD7
	BRA OVER

      
H	MOVFF STARTREG,PORTB
	RRNCF PORTB,F
	GOTO H
	CALL DELAY
	BRA BACK
OVER 
G	MOVFF STARTREG,PORTB
	RLNCF PORTB,F
	GOTO G
	CALL DELAY
	BRA BACK
DELAY
      
    MOVLW D'250'
    MOVWF AB
      
THERE
    NOP
    NOP
      DECF AB,F
      BNZ THERE
      RETURN
     END
